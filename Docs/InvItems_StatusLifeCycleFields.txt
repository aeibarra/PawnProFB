SELECT cast(Created as Date) PCreate, PurchaseDate
FROM InventoryItems
WHERE TransactionNo in (SELECT TransactionNo FROM Transactions WHERE TranType = 'U')




CREATE TRIGGER InventoryItems_BI_BU
BEFORE INSERT, UPDATE
ON InventoryItems
REFERENCING NEW AS n
FOR EACH ROW
BEGIN
  -- Redeemed vs Defaulted: cannot be both
  IF n.RedeemedDate IS NOT NULL AND n.DefaultedDate IS NOT NULL THEN
     RAISERROR 17100 'Item cannot be both Redeemed and Defaulted';
  END IF;

  -- ForSale requires either Defaulted (pawned and forfeited) or Purchase (shop purchase flow)
  IF n.ForSaleDate IS NOT NULL AND n.DefaultedDate IS NULL AND n.PurchaseDate IS NULL THEN
     RAISERROR 17101 'ForSaleDate requires DefaultedDate or PurchaseDate';
  END IF;

  -- Sold requires either Defaulted or Purchase
  IF n.SoldDate IS NOT NULL AND n.DefaultedDate IS NULL AND n.PurchaseDate IS NULL THEN
     RAISERROR 17102 'SoldDate requires DefaultedDate or PurchaseDate';
  END IF;

  -- Melted requires either Defaulted or Purchase
  IF n.MeltedDate IS NOT NULL AND n.DefaultedDate IS NULL AND n.PurchaseDate IS NULL THEN
     RAISERROR 17103 'MeltedDate requires DefaultedDate or PurchaseDate';
  END IF;

  -- (Optional) terminal outcomes must be mutually exclusive
  IF n.SoldDate IS NOT NULL AND n.MeltedDate IS NOT NULL THEN
     RAISERROR 17104 'Item cannot be both Sold and Melted';
  END IF;

END;
